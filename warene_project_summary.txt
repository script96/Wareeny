WARENE (واريني) - PROJECT SUMMARY & HANDOVER DOCUMENT
=======================================================

1. PROJECT NAME AND DESCRIPTION
--------------------------------
Name: Warene (واريني)
Role: AI Assistive Application for Visually Impaired Users in Egypt.
Goal: Provides real-time object detection, Egyptian currency recognition, scene description, and OCR (text reading) in Egyptian Arabic to assist blind users in navigating their environment safely.

2. TECH STACK
--------------
- Framework: Flutter (Dart)
- Vision Models: YOLOv8n (TFLite), Meta Llama 4 Scout (Groq API)
- OCR Engines: Google ML Kit (English), Tesseract OCR (Arabic)
- Speech: flutter_tts (ar-EG)
- Local Processing: flutter_vision (Native TFLite wrapper)
- Training Environment: Python, Ultralytics YOLOv8

3. SYSTEM ARCHITECTURE
-----------------------
- Local Detection: YOLOv8n running locally on the device via TFLite snapshots (every 1.5s).
- Cloud Intelligence: On-demand scene description via Groq API (LLaMA Vision).
- Spatial Engine: Maps 2D coordinates to directional audio (Left, Center, Right) and distance estimation based on bounding box size.
- Preprocessing: Custom center-cropping to square before YOLO inference to preserve aspect ratio.

4. FEATURES & MODULES
----------------------
- 🔍 Real-Time Object Detection (80 COCO classes)
- 💵 Egyptian Currency Recognition (10, 20, 50, 100, 200 EGP)
- 🖼️ Natural Scene Description (Arabic)
- 📖 Hybrid OCR (Arabic & English text reading)
- 🧭 Spatial Awareness (3D-like audio feedback)
- 📷 High-Res Capture (takePicture snapshot mode)

----------------------------
- Groq Cloud API: For Meta Llama 4 Scout Vision model.
- Google ML Kit: For high-speed English/numeric text extraction.

8. IMPORTANT DECISIONS & NOTES
-------------------------------
- Bypassed YUV stride bug in flutter_vision by switching to high-res takePicture snapshots.
- Implemented manual center-cropping to square in Dart to improve model accuracy by preventing distortion.
- Used LTR paragraph direction for Arabic labels to fix alignment issues in the CustomPainter.

9. SETUP & INSTALLATION
------------------------
1. Install Flutter 3.11.0+.
2. Run 'flutter pub get'.
3. Ensure Android NDK is installed for native C++ compilation.
4. Place API keys in gemini_service.dart.
5. Deploy to an Android device (Physical device required for camera/performance).

class _PermissionGateState extends State<PermissionGate> {
  bool _checking = true;

  @override
  void initState() {
    super.initState();
    _requestPermissions();
  }

  Future<void> _requestPermissions() async {
    await [Permission.camera, Permission.microphone].request();
    final cameraOk = await Permission.camera.isGranted;
    final micOk = await Permission.microphone.isGranted;

    if (mounted) {
      if (cameraOk && micOk) {
        Navigator.pushReplacement(
          context,
          MaterialPageRoute(builder: (_) => const CameraScreen()),
        );
      } else {
        setState(() => _checking = false);
      }
    }
  }

  @override
  Widget build(BuildContext context) {
    return Scaffold(
      body: Center(
        child: _checking
            ? const CircularProgressIndicator(color: Colors.white)
            : Column(
                mainAxisAlignment: MainAxisAlignment.center,
                children: [
                  const Icon(Icons.camera_alt, color: Colors.white, size: 64),
                  const SizedBox(height: 24),
                  const Text(
                    'Warene needs camera and\nmicrophone permissions.',
                    textAlign: TextAlign.center,
                    style: TextStyle(color: Colors.white, fontSize: 18),
                  ),
                  const SizedBox(height: 24),
                  ElevatedButton(
                    onPressed: _requestPermissions,
                    child: const Text('Grant Permissions'),
                  ),
                ],
              ),
      ),
    );
  }
}

class _CameraScreenState extends State<CameraScreen>
    with WidgetsBindingObserver {
  CameraController? _controller;
  final FlutterTts _tts = FlutterTts();
  final YoloDetector _detector = YoloDetector();
  final GeminiService _gemini = GeminiService();
  final OCRService _ocrService = OCRService();

  bool _isCameraReady = false;
  bool _isDetecting = false;
  bool _isDescribing = false;
  bool _isOCRProcessing = false;
  OCRLanguage _ocrLanguage = OCRLanguage.english;
  String _statusText = 'Initializing...';
  String _lastSpoken = '';
  DateTime _lastSpeakTime = DateTime.now();

  Timer? _detectionTimer;

  List<Detection> _detections = [];
  int _sourceImageWidth = 1;
  int _sourceImageHeight = 1;
  int _emptyFrameCount = 0;

  @override
  void initState() {
    super.initState();
    WidgetsBinding.instance.addObserver(this);
    WakelockPlus.enable();
    _initTts();
    _initAll();
  }

  @override
  void didChangeAppLifecycleState(AppLifecycleState state) {
    if (_controller == null || !_controller!.value.isInitialized) return;

    if (state == AppLifecycleState.paused) {
      _stopDetection();
      _controller?.dispose();
      _controller = null;
      _isCameraReady = false;
    } else if (state == AppLifecycleState.resumed) {
      _initAll();
    }
  }

  Future<void> _initAll() async {
    await _initCamera();
    await _initDetector(modelType: _detector.currentModel);
    await _gemini.init();
    _startDetectionTimer();
  }

  Future<void> _initTts() async {
    await _tts.setLanguage('ar-EG');
    await _tts.setSpeechRate(0.5);
    await _tts.setVolume(1.0);
    await _tts.speak('مَرْحَبًا، وَرِّينِي جَاهِز');
  }

  Future<void> _initCamera() async {
    if (cameras.isEmpty) {
      if (mounted) setState(() => _statusText = 'No camera found');
      return;
    }

    _controller = CameraController(
      cameras[0],
      ResolutionPreset.high,
      enableAudio: false,
      imageFormatGroup: ImageFormatGroup.jpeg,
    );

    try {
      await _controller!.initialize();
      if (mounted) {
        setState(() {
          _isCameraReady = true;
          _statusText = 'Camera ready';
        });
      }
    } catch (e) {
      if (mounted) setState(() => _statusText = 'Camera error: $e');
    }
  }

  Future<void> _initDetector({DetectionModel modelType = DetectionModel.general}) async {
    if (mounted) setState(() => _statusText = 'Loading AI...');
    try {
      await _detector.loadModel(modelType);
      if (mounted) {
        setState(() => _statusText =
            _detector.isReady ? 'AI Ready ✓' : 'AI Failed');
      }
    } catch (e) {
      if (mounted) setState(() => _statusText = 'AI Error: $e');
    }
  }

  Future<void> _toggleDetectionMode() async {
    if (!_isCameraReady) return;
    _stopDetection();

    if (mounted) {
      setState(() {
        _detections = [];
      });
    }

    final newMode = _detector.currentModel == DetectionModel.general
        ? DetectionModel.currency
        : DetectionModel.general;
    
    if (mounted) {
      setState(() => _statusText = newMode == DetectionModel.general ? 'وضع الأشياء' : 'وضع الفلوس');
    }
    await _tts.speak(newMode == DetectionModel.general ? 'وَضْع الْأَشْيَاء' : 'وَضْع الْفُلُوس');
    
    await _initDetector(modelType: newMode);
    _startDetectionTimer();
  }

  void _startDetectionTimer() {
    if (!_detector.isReady) return;

    _detectionTimer?.cancel();
    _detectionTimer =
        Timer.periodic(const Duration(milliseconds: 1500), (_) {
      _runDetection();
    });
  }

  void _stopDetection() {
    _detectionTimer?.cancel();
    _detectionTimer = null;
  }

  String _getDirection(Detection d, int imageWidth) {
    final centerX = (d.bbox[0] + d.bbox[2]) / 2;
    final third = imageWidth / 3;

    if (centerX < third) {
      return 'عَلَى الْيَسَار';
    } else if (centerX < third * 2) {
      return 'فِي الْوَسَط';
    } else {
      return 'عَلَى الْيَمِين';
    }
  }

  String _getDistance(Detection d, int imageWidth, int imageHeight) {
    final boxW = (d.bbox[2] - d.bbox[0]).abs();
    final boxH = (d.bbox[3] - d.bbox[1]).abs();
    final boxArea = boxW * boxH;
    final imageArea = imageWidth.toDouble() * imageHeight.toDouble();
    final ratio = boxArea / imageArea;

    if (ratio > 0.15) {
      return 'قَرِيب جِدًّا';
    } else if (ratio > 0.05) {
      return 'قَرِيب';
    } else {
      return 'بَعِيد';
    }
  }

  String _buildSpatialDescription(Detection d, int imgW, int imgH) {
    final direction = _getDirection(d, imgW);
    final distance = _getDistance(d, imgW, imgH);
    return '${d.arabicLabel} $direction، $distance';
  }

  Future<void> _triggerOCR() async {
    if (_isOCRProcessing || _isDescribing || !_isCameraReady || _controller == null) return;

    _isOCRProcessing = true;
    _stopDetection();

    if (mounted) {
      setState(() {
        _detections = [];
        _statusText = 'جاري القراءة...';
      });
    }

    HapticFeedback.mediumImpact();
    await _tts.speak('جَارِي الْقِرَاءَة');

    XFile? capturedFile;
    const maxRetries = 2;
    int attempt = 0;

    while (attempt < maxRetries) {
      try {
        capturedFile = await _controller!.takePicture();
        final text = await _ocrService.extractText(capturedFile, language: _ocrLanguage);

        try {
          await File(capturedFile.path).delete();
        } catch (_) {}

        if (mounted) {
          setState(() => _statusText = text);
        }

        if (_ocrLanguage == OCRLanguage.english) {
          await _tts.setLanguage('en-US');
        }

        await _speakAndWait(text);

        if (_ocrLanguage == OCRLanguage.english) {
          await _tts.setLanguage('ar-EG');
        }
        break;
      } catch (e) {
        attempt++;
        if (capturedFile != null) {
          try {
            await File(capturedFile.path).delete();
          } catch (_) {}
        }
        if (attempt == maxRetries) {
          final errorMsg = 'حَصَل خَطَأ فِي الْقِرَاءَة. حَاوِل تَانِي.';
          if (mounted) {
            setState(() => _statusText = errorMsg);
          }
          await _speakAndWait(errorMsg);
        }
      }
    }

    await Future.delayed(const Duration(milliseconds: 500));

    _isOCRProcessing = false;
    if (mounted) setState(() {});
    _startDetectionTimer();
  }

  Future<void> _speakAndWait(String text) async {
    final completer = Completer<void>();
    _tts.setCompletionHandler(() {
      if (!completer.isCompleted) completer.complete();
    });
    _tts.setCancelHandler(() {
      if (!completer.isCompleted) completer.complete();
    });
    _tts.setErrorHandler((msg) {
      if (!completer.isCompleted) completer.complete();
    });
    await _tts.speak(text);
    await completer.future.timeout(
      const Duration(seconds: 30),
      onTimeout: () {},
    );
  }

  Future<void> _runDetection() async {
    if (_isDetecting || _isDescribing || _isOCRProcessing || !_isCameraReady || _controller == null) return;
    _isDetecting = true;

    XFile? capturedFile;
    try {
      capturedFile = await _controller!.takePicture();
      final result = await _detector.detectFromFile(capturedFile.path);

      try {
        await File(capturedFile.path).delete();
      } catch (_) {}

      if (result != null && mounted) {
        if (result.detections.isNotEmpty) {
          _emptyFrameCount = 0;
          setState(() {
            _detections = result.detections;
            _sourceImageWidth = result.imageWidth;
            _sourceImageHeight = result.imageHeight;
          });
        } else {
          _emptyFrameCount++;
          if (_emptyFrameCount >= 3) {
            setState(() {
              _detections = [];
            });
          }
        }

        if (result.detections.isNotEmpty) {
          final now = DateTime.now();
          final timePassed = now.difference(_lastSpeakTime).inSeconds >= 3;

          final best = result.detections.first;
          final spatialDesc = _buildSpatialDescription(
              best, result.imageWidth, result.imageHeight);

          final differentObject = spatialDesc != _lastSpoken;

          if (differentObject || timePassed) {
            _lastSpoken = spatialDesc;
            _lastSpeakTime = now;

            if (mounted) {
              setState(() => _statusText =
                  '${best.arabicLabel} ${_getDirection(best, result.imageWidth)} '
                  '(${(best.confidence * 100).toStringAsFixed(0)}%)');
            }

            final topDetections = result.detections.take(3);
            final phrases = topDetections.map((d) =>
                _buildSpatialDescription(
                    d, result.imageWidth, result.imageHeight));
            final ttsText = phrases.join('، ');
            await _tts.speak(ttsText);
          }
        }
      }
    } catch (e) {
      if (capturedFile != null) {
        try {
          await File(capturedFile.path).delete();
        } catch (_) {}
      }
    } finally {
      _isDetecting = false;
    }
  }

  Future<void> _describeScene() async {
    if (_isDescribing || !_isCameraReady || _controller == null) return;

    _isDescribing = true;
    _stopDetection();

    if (mounted) {
      setState(() {
        _detections = [];
        _statusText = 'جاري الوصف...';
      });
    }

    HapticFeedback.mediumImpact();
    await _tts.speak('جَارِي الْوَصْف');

    XFile? capturedFile;
    try {
      capturedFile = await _controller!.takePicture();
      final imageBytes = await File(capturedFile.path).readAsBytes();

      try {
        await File(capturedFile.path).delete();
      } catch (_) {}

      final description = await _gemini.describeScene(imageBytes);

      if (mounted) {
        setState(() => _statusText = description);
      }
      
      await _speakAndWait(description);
      await Future.delayed(const Duration(milliseconds: 500));
    } catch (e) {
      final errorMsg = 'حَصَل خَطَأ فِى الْوَصْف. حَاوِل تَانِي.';
      if (mounted) {
        setState(() => _statusText = errorMsg);
      }
      await _speakAndWait(errorMsg);
      if (capturedFile != null) {
        try {
          await File(capturedFile.path).delete();
        } catch (_) {}
      }
    } finally {
      _isDescribing = false;
      if (mounted) setState(() {});
      _startDetectionTimer();
    }
  }

  @override
  void dispose() {
    WidgetsBinding.instance.removeObserver(this);
    WakelockPlus.disable();
    _stopDetection();
    _controller?.dispose();
    _tts.stop();
    _detector.dispose();
    _gemini.dispose();
    _ocrService.dispose();
    super.dispose();
  }

class BoundingBoxPainter extends CustomPainter {
  final List<Detection> detections;
  final int sourceWidth;
  final int sourceHeight;
  final double displayWidth;
  final double displayHeight;
  final double offsetX;
  final double offsetY;

  static const List<Color> _boxColors = [
    Color(0xFF00FF00),
    Color(0xFF00BFFF),
    Color(0xFFFF6347),
    Color(0xFFFFD700),
    Color(0xFFFF69B4),
    Color(0xFF7B68EE),
    Color(0xFFFF8C00),
    Color(0xFF00CED1),
  ];

  BoundingBoxPainter({
    required this.detections,
    required this.sourceWidth,
    required this.sourceHeight,
    required this.displayWidth,
    required this.displayHeight,
    required this.offsetX,
    required this.offsetY,
  });

  @override
  void paint(Canvas canvas, Size size) {
    final scaleX = displayWidth / sourceWidth;
    final scaleY = displayHeight / sourceHeight;

    for (int i = 0; i < detections.length; i++) {
      final d = detections[i];
      final color = _boxColors[i % _boxColors.length];

      final left = d.bbox[0] * scaleX + offsetX;
      final top = d.bbox[1] * scaleY + offsetY;
      final right = d.bbox[2] * scaleX + offsetX;
      final bottom = d.bbox[3] * scaleY + offsetY;

      final rect = Rect.fromLTRB(left, top, right, bottom);

      final boxPaint = Paint()
        ..color = color
        ..style = PaintingStyle.stroke
        ..strokeWidth = 2.5;
      canvas.drawRect(rect, boxPaint);

      final fillPaint = Paint()
        ..color = color.withAlpha(30)
        ..style = PaintingStyle.fill;
      canvas.drawRect(rect, fillPaint);

      final label =
          '${d.arabicLabel} ${(d.confidence * 100).toStringAsFixed(0)}%';

      final textStyle = ui.TextStyle(
        color: Colors.white,
        fontSize: 13,
        fontWeight: FontWeight.bold,
      );

      final paragraphBuilder = ui.ParagraphBuilder(
        ui.ParagraphStyle(
          textDirection: ui.TextDirection.ltr,
          fontSize: 13,
          maxLines: 1,
        ),
      )
        ..pushStyle(textStyle)
        ..addText(label);

      final paragraph = paragraphBuilder.build();
      paragraph.layout(ui.ParagraphConstraints(width: (right - left).clamp(80, 300)));

      final textWidth = paragraph.longestLine + 10;
      final textHeight = paragraph.height + 6;

      final labelTop = (top - textHeight - 2).clamp(0.0, size.height - textHeight);

      final labelBg = Rect.fromLTWH(left, labelTop, textWidth, textHeight);
      canvas.drawRRect(
        RRect.fromRectAndRadius(labelBg, const Radius.circular(4)),
        Paint()..color = color.withAlpha(200),
      );

      canvas.drawParagraph(paragraph, Offset(left + 5, labelTop + 3));
    }
  }

  @override
  bool shouldRepaint(covariant BoundingBoxPainter oldDelegate) {
    return detections != oldDelegate.detections;
  }
}

  Future<Uint8List?> _centerCropToSquare(Uint8List imageBytes) async {
    try {
      final codec = await ui.instantiateImageCodec(imageBytes);
      final frame = await codec.getNextFrame();
      final srcImage = frame.image;
      final w = srcImage.width;
      final h = srcImage.height;

      if ((w - h).abs() < 10) {
        srcImage.dispose();
        codec.dispose();
        return null;
      }

      final side = math.min(w, h);
      final cropX = (w - side) / 2;
      final cropY = (h - side) / 2;

      final recorder = ui.PictureRecorder();
      final canvas = ui.Canvas(recorder);
      canvas.drawImageRect(
        srcImage,
        ui.Rect.fromLTWH(cropX, cropY, side.toDouble(), side.toDouble()),
        ui.Rect.fromLTWH(0, 0, side.toDouble(), side.toDouble()),
        ui.Paint(),
      );
      final picture = recorder.endRecording();
      final croppedImage = await picture.toImage(side, side);

      final byteData =
          await croppedImage.toByteData(format: ui.ImageByteFormat.png);

      srcImage.dispose();
      croppedImage.dispose();
      picture.dispose();
      codec.dispose();

      return byteData?.buffer.asUint8List();
    } catch (e) {
      return null;
    }

  Future<DetectionResult?> detectFromFile(String imagePath) async {
    if (!_isReady) return null;

    try {
      final Uint8List originalBytes = await File(imagePath).readAsBytes();

      final origCodec = await ui.instantiateImageCodec(originalBytes);
      final origFrame = await origCodec.getNextFrame();
      final int origW = origFrame.image.width;
      final int origH = origFrame.image.height;
      origFrame.image.dispose();
      origCodec.dispose();

      final int side = math.min(origW, origH);
      final double cropX;
      final double cropY;

      Uint8List inferenceBytes;
      int inferenceW, inferenceH;

      if (_currentModel == DetectionModel.currency) {
        inferenceBytes = originalBytes;
        inferenceW = origW;
        inferenceH = origH;
        cropX = 0;
        cropY = 0;
      } else {
        cropX = (origW - side) / 2;
        cropY = (origH - side) / 2;
        final cropped = await _centerCropToSquare(originalBytes);
        if (cropped != null) {
          inferenceBytes = cropped;
          inferenceW = side;
          inferenceH = side;
        } else {
          inferenceBytes = originalBytes;
          inferenceW = origW;
          inferenceH = origH;
        }
      }

      final confThresh = _currentModel == DetectionModel.currency ? 0.15 : 0.25;
      final iouThresh = _currentModel == DetectionModel.currency ? 0.3 : 0.4;
      final results = await _vision.yoloOnImage(
        bytesList: inferenceBytes,
        imageHeight: inferenceH,
        imageWidth: inferenceW,
        iouThreshold: iouThresh,
        confThreshold: confThresh,
        classThreshold: confThresh,
      );

      if (results.isEmpty) {
        return DetectionResult(
          detections: [],
          imageWidth: origW,
          imageHeight: origH,
        );
      }

      final double minBoxArea = _currentModel == DetectionModel.currency
          ? origW * origH * 0.002
          : origW * origH * 0.01;

      final detections = <Detection>[];
      for (final result in results) {
        final String tag = result['tag'] as String;
        final List<dynamic> box = result['box'] as List<dynamic>;
        final double confidence = (box[4] as num).toDouble();

        final double x1 = (box[0] as num).toDouble() + cropX;
        final double y1 = (box[1] as num).toDouble() + cropY;
        final double x2 = (box[2] as num).toDouble() + cropX;
        final double y2 = (box[3] as num).toDouble() + cropY;

        final double boxArea = (x2 - x1).abs() * (y2 - y1).abs();
        if (boxArea < minBoxArea) continue;

        String arLabel = tag;
        if (_currentModel == DetectionModel.general) {
          arLabel = arabicNames[tag] ?? tag;
        } else {
          arLabel = currencyArabicNames[tag] ?? tag;
        }

        detections.add(Detection(
          label: tag,
          arabicLabel: arLabel,
          confidence: confidence,
          bbox: [x1, y1, x2, y2],
        ));
      }

      detections.sort((a, b) => b.confidence.compareTo(a.confidence));

      return DetectionResult(
        detections: detections,
        imageWidth: origW,
        imageHeight: origH,
      );
    } catch (e) {
      return null;
    }

    final imagePath = imageFile.path;
    String rawText = '';

        final inputImage = InputImage.fromFilePath(imagePath);
        final RecognizedText result = await _latinRecognizer.processImage(inputImage);
        rawText = result.text.trim();

          imagePath,
          config: const OCRConfig(language: 'ara', engine: OCREngine.tesseract),
        );
        rawText = rawText.trim();
      } catch (e) {}

    _latinRecognizer.close();
  }
